import React, { useState } from "react";
import { View, Text, TextInput, TouchableOpacity, StyleSheet, ScrollView } from "react-native";

export default function App() {
  const [foodInput, setFoodInput] = useState("");
  const [result, setResult] = useState(null);
  const [loading, setLoading] = useState(false);

  const analyzeMood = async () => {
    setLoading(true);

    // ✨ تحليل بسيط محلي (بدون API في البداية)
    let mood = "";
    let advice = "";

    if (foodInput.includes("pizza") || foodInput.includes("burger") || foodInput.includes("بطاطس")) {
      mood = "😴 خمول وراحة مؤقتة";
      advice = "جرب تضيف خضار أو فواكه لتحسين طاقتك.";
    } else if (foodInput.includes("salad") || foodInput.includes("fruit") || foodInput.includes("تفاح")) {
      mood = "😊 طاقة جيدة ومزاج إيجابي!";
      advice = "استمر على الأكل الخفيف والمغذي!";
    } else {
      mood = "🤔 صعب التحديد بدقة";
      advice = "جرب تكتب أكلك بشكل أوضح (مثلاً: بيتزا، رز، سمك).";
    }

    setResult({ mood, advice });
    setLoading(false);
  };

  return (
    <ScrollView contentContainerStyle={styles.container}>
      <Text style={styles.title}>🍽️ Mood & Food AI</Text>
      <Text style={styles.subtitle}>اكتب وش أكلت اليوم، وخلي الذكاء الصناعي يحلل مزاجك 👇</Text>

      <TextInput
        style={styles.input}
        placeholder="مثلاً: بيتزا وبطاطس"
        value={foodInput}
        onChangeText={setFoodInput}
      />

      <TouchableOpacity style={styles.button} onPress={analyzeMood}>
        <Text style={styles.buttonText}>{loading ? "جاري التحليل..." : "حلل المزاج"}</Text>
      </TouchableOpacity>

      {result && (
        <View style={styles.resultBox}>
          <Text style={styles.resultTitle}>تحليل المزاج:</Text>
          <Text style={styles.resultText}>{result.mood}</Text>
          <Text style={styles.resultTitle}>نصيحة غذائية:</Text>
          <Text style={styles.resultText}>{result.advice}</Text>
        </View>
      )}
    </ScrollView>
  );
}

const styles = StyleSheet.create({
  container: {
    flexGrow: 1,
    alignItems: "center",
    justifyContent: "center",
    padding: 20,
    backgroundColor: "#fff",
  },
  title: {
    fontSize: 28,
    fontWeight: "bold",
    marginTop: 50,
  },
  subtitle: {
    fontSize: 16,
    color: "#666",
    textAlign: "center",
    marginVertical: 20,
  },
  input: {
    width: "100%",
    borderWidth: 1,
    borderColor: "#ccc",
    borderRadius: 10,
    padding: 15,
    fontSize: 16,
    marginBottom: 20,
  },
  button: {
    backgroundColor: "#00A86B",
    paddingVertical: 15,
    paddingHorizontal: 30,
    borderRadius: 10,
  },
  buttonText: {
    color: "white",
    fontSize: 18,
  },
  resultBox: {
    backgroundColor: "#f2f2f2",
    marginTop: 30,
    padding: 20,
    borderRadius: 15,
    width: "100%",
  },
  resultTitle: {
    fontSize: 18,
    fontWeight: "bold",
    marginTop: 10,
  },
  resultText: {
    fontSize: 16,
    marginTop: 5,
  },
});
